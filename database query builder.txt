 **join tables
  posts table   categories table

  DB::table( 'posts')->join( 'categories', 'posts.category_id', '=', 'categories.id')->get();	

 * you can specfiy the data that will show
  DB::table( 'posts')->join( 'categories', 'posts.category_id', '=', 'categories.id')
  ->select('posts.*')
  ->get();
