 Schema::create('comments', function (Blueprint $table) {
            $table->id(); // Creates an auto-incrementing primary key column named `id`
            $table->text('body'); // Stores the comment text
            $table->unsignedBigInteger('commentable_id'); // Stores the ID of the related model (e.g., Post or Video)
            $table->string('commentable_type'); // Stores the class name of the related model (e.g., App\Models\Post or App\Models\Video)
